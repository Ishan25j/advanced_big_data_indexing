# ============================================
# RANGE QUERIES
# ============================================

# 20. Search plans created after a specific date
GET /healthcare_plans/_search
{
  "query": {
    "bool": {
      "must": [
        { "term": { "objectType": "plan" } },
        {
          "range": {
            "creationDate": {
              "gte": "01-01-2024",
              "format": "dd-MM-yyyy"
            }
          }
        }
      ]
    }
  }
}

# 21. Find cost shares with deductible between 500 and 2000
GET /healthcare_plans/_search
{
  "query": {
    "bool": {
      "must": [
        { "term": { "objectType": "membercostshare" } },
        {
          "range": {
            "deductible": {
              "gte": 500,
              "lte": 2000
            }
          }
        }
      ]
    }
  }
}

  }

# 23. Plans that have high deductible cost shares (>= 1000)
GET /healthcare_plans/_search
{
  "query": {
    "bool": {
      "must": [
        { "term": { "objectType": "plan" } }
      ],
      "filter": [
        {
          "has_child": {
            "type": "planCostShares",
            "query": {
              "range": {
                "deductible": { "gte": 1000 }
              }
            }
          }
        }
      ]
    }
  }
}
